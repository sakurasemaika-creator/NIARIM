```python
"""Restore the standalone S38 parts into a new owned directory without an SDK."""
from pathlib import Path
import argparse, hashlib, json, zipfile
PARTS = [
    ('1cwKG-bLeICkyUD3Q4w98mxOAOQkIZO1K', '1d53fba400ba8256e4eec2f7769e55bf14fefab5f889a79785e9a4517abbc220'),
    ('1IVKKE3_w0st5KAwn-qxAvOpzUEXZX0ip', 'eca842182aacfe2500ccaeed42c67e322700a89bb54889ad64ddbe77c13a24cd'),
]
if __name__ == '__main__':
    p = argparse.ArgumentParser(__doc__)
    p.add_argument('--archives', type=Path, nargs=2, required=True)
    p.add_argument('--output', type=Path, required=True)
    args = p.parse_args()
    assert not args.output.exists() and not args.output.is_symlink()
    expected = None
    extracted = {}
    for archive, (drive_id, digest) in zip(args.archives, PARTS):
        with archive.open('rb') as f:
            assert hashlib.file_digest(f, 'sha256').hexdigest() == digest, drive_id
        with zipfile.ZipFile(archive) as z:
            assert z.testzip() is None
            m = json.loads(z.read('MANIFEST.json'))
            modes = json.loads(z.read('FILE_METADATA.json'))
            if expected is None: expected = m['target_bundle_sha256']
            assert expected == m['target_bundle_sha256']
            for name, sha in m['sha256'].items():
                prefix = 'NIARIM/build/linux/x64/debug/bundle/'
                assert name.startswith(prefix)
                rel = Path(name[len(prefix):])
                assert not rel.is_absolute() and '..' not in rel.parts
                data = z.read(name)
                assert hashlib.sha256(data).hexdigest() == sha == expected[name]
                dst = args.output / rel
                dst.parent.mkdir(parents=True, exist_ok=True)
                dst.write_bytes(data)
                dst.chmod(modes[name]['mode'])
                extracted[name] = sha
    assert extracted == expected
    assert m['kernel_sha256'] == '950710eb7dce8765d7954dc7f95f6f1e6227a62872886e4ab1aed205066ae5ce'
    print(json.dumps({'files':len(extracted),'kernel_sha256':m['kernel_sha256'],'requires_SDK':False,'requires_older_runtime':False}))
```
